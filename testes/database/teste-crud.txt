-- TESTE 1 — DeveCriarNovoRegistroComDadosValidos
-- Função: CriarRegistro(registro)

INSERT INTO produto (nome, preco, quantidade)
    VALUES ('Biscoito de Arroz', 4.50, 60);

SELECT CASE
    WHEN COUNT(*) = 1
    THEN 'PASSOU — DeveCriarNovoRegistroComDadosValidos'
    ELSE 'FALHOU — DeveCriarNovoRegistroComDadosValidos'
END AS resultado
FROM produto
WHERE nome = 'Biscoito de Arroz'
  AND preco = 4.50
  AND quantidade = 60;

   SET preco = 5.00,
       descricao = 'Pacote 200g premium',
       atualizado_em = NOW()
 WHERE id = 1;

    WHEN preco = 5.00
     AND descricao = 'Pacote 200g premium'

FROM produto

-- TESTE 7 — DeveRetornarErroAoAtualizarRegistroInexistente
-- Função: AtualizarRegistro(registro)

DO $$
DECLARE
    linhas INTEGER;
BEGIN
    UPDATE produto SET preco = 99.99 WHERE id = 9999;
    GET DIAGNOSTICS linhas = ROW_COUNT;
    IF linhas = 0 THEN
        RAISE NOTICE 'PASSOU — DeveRetornarErroAoAtualizarRegistroInexistente';
    ELSE
        RAISE NOTICE 'FALHOU — DeveRetornarErroAoAtualizarRegistroInexistente';
    END IF;
END;
$$;
