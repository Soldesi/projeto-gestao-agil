-- TESTE 1 — DeveAdicionarProdutoAoEstoque
-- Função: AdicionarProduto(produto)

INSERT INTO produto (nome, preco, quantidade, quantidade_minima)
    VALUES ('Molho de Soja', 5.90, 50, 10);

SELECT CASE
    WHEN COUNT(*) = 1
    THEN 'PASSOU — DeveAdicionarProdutoAoEstoque'
    ELSE 'FALHOU — DeveAdicionarProdutoAoEstoque'
END AS resultado
FROM produto
WHERE nome = 'Molho de Soja'
  AND preco = 5.90
  AND quantidade = 50;

-- TESTE 2 — DeveAumentarQuantidadeAoEntrarItens
-- Função: AplicarEntradaItem(item)

INSERT INTO movimentacao_estoque (produto_id, tipo, quantidade, observacao)
    VALUES (1, 'ENTRADA', 20, 'Reposição de lote');

UPDATE produto
   SET quantidade = quantidade + 20,
       atualizado_em = NOW()
 WHERE id = 1;

SELECT CASE
    WHEN p.quantidade = 70
     AND m.tipo = 'ENTRADA'
     AND m.quantidade = 20
    THEN 'PASSOU — DeveAumentarQuantidadeAoEntrarItens'
    ELSE 'FALHOU — DeveAumentarQuantidadeAoEntrarItens'
END AS resultado
FROM produto p
JOIN movimentacao_estoque m ON m.produto_id = p.id
WHERE p.id = 1 AND m.tipo = 'ENTRADA';
