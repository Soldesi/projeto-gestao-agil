-- TESTE 1 — DeveAceitarPagamento
-- Função: RealizarPagamento(pagamento)

INSERT INTO movimentacao_caixa (tipo, valor, descricao, categoria)
    VALUES ('ENTRADA', 50.00, 'Venda de produtos', 'VENDAS');

SELECT CASE
    WHEN COUNT(*) = 1
    THEN 'PASSOU — DeveAceitarPagamento'
    ELSE 'FALHOU — DeveAceitarPagamento'
END AS resultado
FROM movimentacao_caixa
WHERE tipo = 'ENTRADA' AND valor = 50.00 AND categoria = 'VENDAS';

-- TESTE 3 — DeveRealizarTransacaoBancaria
-- Função: ProcessarTransacao(transacao)

INSERT INTO movimentacao_caixa (tipo, valor, descricao, categoria)
    VALUES ('ENTRADA', 150.00, 'Transferência bancária', 'TRANSFERENCIA');

SELECT CASE
    WHEN SUM(CASE WHEN tipo='ENTRADA' THEN valor ELSE 0 END)
       - SUM(CASE WHEN tipo='SAIDA'   THEN valor ELSE 0 END) = 200.00
    THEN 'PASSOU — DeveRealizarTransacaoBancaria (saldo = R$ 200,00)'
    ELSE 'FALHOU — DeveRealizarTransacaoBancaria'
END AS resultado
FROM movimentacao_caixa;

    VALUES ('SAIDA', 25.00, 'Cancelamento Venda #1', 'CANCELAMENTO');

UPDATE produto SET quantidade = quantidade + 10 WHERE id = 1;

SELECT CASE
    WHEN v.status = 'CANCELADA' AND p.quantidade = 110
    THEN 'PASSOU — DeveAceitarCancelamentoDeVenda'
    ELSE 'FALHOU — DeveAceitarCancelamentoDeVenda'

-- TESTE 6 — DeveLiberarParcelamento
-- Função: AplicarParcelamento(venda)

INSERT INTO venda (status, valor_total, parcelado) VALUES ('ATIVA', 150.00, 3);

SELECT CASE
    WHEN parcelado = 3
     AND (valor_total / parcelado) >= 20.00
    THEN 'PASSOU — DeveLiberarParcelamento'
    ELSE 'FALHOU — DeveLiberarParcelamento'
END AS resultado
FROM venda
WHERE id = 2;
