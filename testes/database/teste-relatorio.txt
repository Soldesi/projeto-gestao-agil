-- TESTE 6 — DeveCalcularTotaisCorretamente
-- Função: CalcularTotais(relatorio)

SELECT CASE
    WHEN total_entradas = 730.00
     AND total_saidas   = 120.00
     AND saldo_final    = 610.00
    THEN 'PASSOU — DeveCalcularTotaisCorretamente'
    ELSE 'FALHOU — DeveCalcularTotaisCorretamente'
END AS resultado
FROM (
    SELECT
        SUM(CASE WHEN tipo = 'ENTRADA' THEN valor ELSE 0    END) AS total_entradas,
        SUM(CASE WHEN tipo = 'SAIDA'   THEN valor ELSE 0    END) AS total_saidas,
        SUM(CASE WHEN tipo = 'ENTRADA' THEN valor ELSE -valor END) AS saldo_final
    FROM movimentacao_caixa
) totais;

-- TESTE 7 — DeveExportarRelatorioEmPdf
-- Função: ExportarRelatorio(relatorio)

SELECT CASE
    WHEN COUNT(*) >= 1
    THEN 'PASSOU — DeveExportarRelatorioEmPdf (dados disponíveis para exportação)'
    ELSE 'FALHOU — DeveExportarRelatorioEmPdf'
END AS resultado
FROM (
    SELECT
        p.nome,
        SUM(CASE WHEN me.tipo='ENTRADA' THEN me.quantidade ELSE 0 END) AS total_entrada,
        SUM(CASE WHEN me.tipo='SAIDA'   THEN me.quantidade ELSE 0 END) AS total_saida,
        p.quantidade AS saldo_atual
    FROM produto p
    LEFT JOIN movimentacao_estoque me ON me.produto_id = p.id
    WHERE p.ativo = TRUE
    GROUP BY p.id, p.nome, p.quantidade
) relatorio_estoque;
